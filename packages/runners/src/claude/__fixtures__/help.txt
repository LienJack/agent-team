Usage: claude [options] [command] [prompt]

Options:
  --allow-dangerously-skip-permissions  (listed by the CLI; never passed)
  --allowedTools, --allowed-tools <tools...>  allow tools
  --dangerously-skip-permissions        (listed by the CLI; never passed)
  --disallowedTools, --disallowed-tools <tools...>  deny tools
  --effort <level>                      effort level
  --json-schema <schema>                structured output schema
  --mcp-config <configs...>             MCP servers
  --model <model>                       model
  --output-format <format>              text, json, stream-json
  --permission-mode <mode>              permission mode
  --permission-prompts <target>         host or none
  -p, --print                           print and exit
  --restricted                          no user/project settings, file tools confined
  -r, --resume [value]                  resume a session
  --safe-mode                           no customizations
  --settings <file-or-json>             more settings
  --strict-mcp-config                   only --mcp-config servers
  --tools <tools...>                    built-in tools
  --verbose                             verbose
  -v, --version                         version
